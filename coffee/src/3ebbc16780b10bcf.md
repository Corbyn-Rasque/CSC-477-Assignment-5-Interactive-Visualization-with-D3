```js
class Globe {
  constructor(data, containing, svg) {
    this.data       = data;
    this.containing = containing;
    this.svg        = svg;
    this.hovered    = null;
    this.selected   = null;

    this.projection = d3.geoOrthographic()
                        .scale(chart.globe.radius)
                        .translate([0, 0])
                        .rotate(chart.globe.start)
                        .clipAngle(90);

    this.path = d3.geoPath().projection(this.projection);
    this.view = this.containing.append('g').attr('clip-path', 'url(#globe-clip)');

    this.addOcean();
    this.addGraticule();
    this.addCountries();
    this.addBorders();
    this.addAtmosphere();

    this.selection   = false;
    this._reticuleEl = null;
    this._onSelect   = () => {};

    this._tooltip = d3.select("body").append("div")
      .style("position", "absolute")
      .style("background", "rgba(0,0,0,0.75)")
      .style("color", "#fff")
      .style("padding", "4px 8px")
      .style("border-radius", "4px")
      .style("font-size", "12px")
      .style("pointer-events", "none")
      .style("opacity", 0);
  }

  addOcean() {
    this.view.append('circle')
             .attr('cx', 0).attr('cy', 0)
             .attr('r', chart.globe.radius)
             .attr('fill', chart.globe.colors.ocean);
  }

  addGraticule() {
    this.graticule = this.view.append('path')
                              .datum(d3.geoGraticule()())
                              .attr('fill', 'none')
                              .attr('stroke', chart.globe.colors.graticule)
                              .attr('stroke-width', chart.globe.stroke.graticule);
  }

  addCountries() {
    this.countries = this.view.append('g');
  }

  addBorders() {
    this.borders = this.view.append('path')
                            .attr('fill', 'none')
                            .attr('stroke', chart.globe.colors.borders)
                            .attr('stroke-width', chart.globe.stroke.borders);
  }

  addAtmosphere() {
    this.containing.append('circle')
                   .attr('cx', 0).attr('cy', 0)
                   .attr('r', chart.globe.radius + chart.globe.stroke.outline / 2)
                   .attr('fill', 'none')
                   .attr('stroke', chart.globe.colors.atmosphere)
                   .attr('stroke-width', chart.globe.stroke.outline)
                   .attr('stroke-opacity', chart.globe.stroke.opacity);
  }

  reticule(k = 2) {
    if (!this.selection) return;
    if (!this._reticuleEl) {
      this._reticuleEl = this.containing.append('circle')
        .attr('cx', 0)
        .attr('cy', 0)
        .attr('fill', 'none')
        .attr('stroke', '#CC0000')
        .attr('stroke-width', chart.globe.stroke.outline)
        .attr('stroke-opacity', chart.globe.stroke.opacity);
    }
    const base = chart.globe.radius + chart.globe.stroke.outline / 2;
    this._reticuleEl.attr('r', base * 0.5 * k);
  }

  contained() {
    if (!this.selection || !this._reticuleEl) return [];

    const k    = this.svg.node().__zoom?.k ?? 2;
    const base = chart.globe.radius + chart.globe.stroke.outline / 2;
    const r    = base * 0.5 * k;
    const cx = 0;
    const cy = 0;

    return GLOBE.countries.features.filter(d => {
      if (!this.visible(d)) return false;
      const centroid = this.path.centroid(d);
      if (isNaN(centroid[0]) || isNaN(centroid[1])) return false;
      const dx = centroid[0] - cx;
      const dy = centroid[1] - cy;
      return Math.sqrt(dx * dx + dy * dy) <= r;
    }).map(d => d.properties.name);
  }

  visible(feature) {
    if (!feature.geometry) return false;
    const center = this.projection.invert([0, 0]);

    function coordinates(polygon) {
      const coords = feature.geometry.type === 'MultiPolygon' ? polygon[0] : polygon;
      return coords.some(coord => d3.geoDistance(coord, center) < Math.PI / 2);
    }

    return feature.geometry.coordinates.some(coordinates);
  }

  hasDatum(d) {
    return Array.isArray(this.data) &&
      this.data.some(item => item.origin === d.properties.name);
  }

  scroll() {
    this.svg.node().addEventListener("wheel", evt => {
      if (evt.ctrlKey) return;
      evt.preventDefault();
      chart.globe.start[0] -= evt.deltaX * 0.3;
      chart.globe.start[1] += evt.deltaY * 0.3;
      chart.globe.start[1] = Math.max(-90, Math.min(90, chart.globe.start[1]));
      this.projection.rotate(chart.globe.start);
      this.render();
    }, { passive: false });
  }

  zoom() {
    this._prevK = 1;
  
    this.zooming = d3.zoom()
      .scaleExtent([0.1, 2])
      .filter(evt => {
        if (evt.type === "dblclick") return false;
        if (evt.type === "mousedown") return false;
        if (evt.type === "wheel" && !evt.ctrlKey) return false;
        if (evt.target.closest(".scaa-wheel")) return false; // ignore SCAA clicks
        return true;
      })
      .on("zoom", evt => {
        const k = evt.transform.k;
        const pinchingIn  = k < this._prevK;
        const pinchingOut = k > this._prevK;
  
        if (!this.selection && !this.selected && pinchingIn) {
          this.selection = true;
          this.svg.call(this.zooming.transform, d3.zoomIdentity.scale(2));
          this._prevK = 2;
          this.render();
          return;
        }
  
        if (this.selection && pinchingOut && k >= 2) {
          this.selection = false;
          if (this._reticuleEl) {
            this._reticuleEl.remove();
            this._reticuleEl = null;
          }
          this.svg.call(this.zooming.transform, d3.zoomIdentity.scale(2));
          this._prevK = 2;
          this.render();
          return;
        }
  
        this._prevK = k;
        this.render();
      });
  
    this.svg.call(this.zooming);
  }

  interactivity({ scroll = true, zoom = true, reset = true, onReset, onSelect } = {}) {
    this._onSelect = onSelect ?? (() => {});
    if (scroll) this.scroll();
    if (zoom)   this.zoom();
    if (reset)  this.reset(onReset);
  }

  reset(onReset) {
    this.svg.node().addEventListener("dblclick", () => {
      this.selected  = null;
      this.selection = false;
      if (this._reticuleEl) {
        this._reticuleEl.remove();
        this._reticuleEl = null;
      }
      if (onReset) onReset();
      this._onSelect(null);
      this.render();
    });
  }

  render(onHover) {
    this._onHover = onHover || this._onHover || (() => {});
    const k = this.svg.node().__zoom?.k ?? 2;
    this.reticule(k);
  
    this.countries.selectAll('path')
      .data(GLOBE.countries.features)
      .join(
        enter => enter.append('path')
          .on('mouseover', (evt, d) => {
            if (!this.hasDatum(d)) return;
            this.hovered = d;
  
            this._tooltip
              .style("opacity", 1)
              .html(d.properties.name);
  
            this.render();
            this._onHover(d);
          })
          .on('mousemove', (evt) => {
            this._tooltip
              .style("left", (evt.pageX + 12) + "px")
              .style("top",  (evt.pageY - 28) + "px");
          })
          .on('mouseout', () => {
            this.hovered = null;
            this._tooltip.style("opacity", 0);
            this.render();
            this._onHover(null);
          })
          .on('click', (evt, d) => {
            if (!this.hasDatum(d)) return;
            this.selected  = d;
            this.selection = false;
            if (this._reticuleEl) {
              this._reticuleEl.remove();
              this._reticuleEl = null;
            }
            this._tooltip.style("opacity", 0);
            this._onSelect(d.properties.name);
            this.render();
          })
      )
      .attr('d', d => this.path(d))
      .attr('fill', d => {
        const match = Array.isArray(this.data) &&
          this.data.find(item => item.origin === d.properties.name);
      
        return match ? match.color : chart.globe.colors.land;
      })
      .attr('stroke', d => {
        if (!this.hasDatum(d)) return 'none';
        if (d === this.selected) return chart.globe.colors.selected;
        if (d === this.hovered)  return chart.globe.colors.hovered;
        return 'none';
      })
      .attr('stroke-width', d => d === this.selected ? 2 : 1.5)
      .attr('cursor', d => this.hasDatum(d) ? 'pointer' : 'default');
      // .attr('stroke-width', d => d === this.selected ? 2 : 1.5);

    this.graticule.attr('d', this.path(d3.geoGraticule()()));
    this.borders.attr('d', this.path(GLOBE.borders));
    this._onHover(this.hovered);
  }
}
```

```js
class SCAA {
  constructor(data, svg, coffee = []) {
    this.data   = data;
    this.svg    = svg;
    this.coffee = coffee;
    this.path   = svg.append("g");
    this.text   = svg.append("g");
    this.root   = d3.hierarchy(this.data)
      .sum(d => d.children ? 0 : 1);
  }

  globeData(coffee) {
    function tagColorMap(node, color = null) {
      const c = node.colour ?? color;
      return [[node.name, c], ...(node.children ?? []).flatMap(child => tagColorMap(child, c))];
    }
    const scaaColors = new Map(this.data.children.flatMap(child => tagColorMap(child)));

    const remapped = Array.from(
      d3.rollups(
        coffee,
        rows => {
          const tagCounts = d3.rollups(rows, v => v.length, d => d.tag)
            .filter(([tag]) => scaaColors.has(tag))
            .sort((a, b) => b[1] - a[1]);
          return tagCounts.length ? scaaColors.get(tagCounts[0][0]) : null;
        },
        d => d.origin
      ),
      ([origin, color]) => ({ origin, color })
    ).filter(d => d.origin !== null);

    const deduped = new Map();
    for (const d of remapped) {
      if (!deduped.has(d.origin) || (d.color && !deduped.get(d.origin))) {
        deduped.set(d.origin, d.color);
      }
    }

    return Array.from(deduped, ([origin, color]) => ({
      origin,
      color: color ?? chart.globe.colors.land
    }));
  }

  radii(depth) {
    const radius = chart.radius;
    const inner  = chart.rings.thickness.inner * radius;
    const gap    = chart.rings.gap * radius;
    const scaled = chart.rings.thickness.rings.map(t => t * radius - gap);
    const cumulative = [inner];
    for (const thickness of scaled) {
      cumulative.push(cumulative[cumulative.length - 1] + thickness + gap);
    }
    return [cumulative[depth - 1], cumulative[depth] - gap];
  }

  render() {
    d3.partition()
      .size([2 * Math.PI, chart.radius * chart.radius])(this.root);

    this.root.each(d => {
      d.original = { x0: d.x0, x1: d.x1 };
      d.current  = { x0: d.x0, x1: d.x1, depth: d.depth };
    });

    this.arc = d3.arc()
      .startAngle(d => d.x0)
      .endAngle(d => d.x1)
      .padAngle(chart.rings.gap)
      .innerRadius(d => this.radii(d.depth)[0])
      .outerRadius(d => this.radii(d.depth)[1]);

    this.paths = this.path.selectAll("path")
      .data(this.root.descendants().filter(d => d.depth))
      .join("path")
      .attr("fill", d => d.data.colour ?? d.parent?.data.colour ?? "#ccc")
      .attr("fill-opacity", d => this._arcVisible(d.current) ? 1 : 0)
      .attr("pointer-events", d => this._arcVisible(d.current) ? "auto" : "none")
      .attr("d", d => this.arc(d.current));

    this.paths.filter(d => d.children)
      .style("cursor", "pointer")
      .on("click", (event, p) => this._clicked(event, p));

    this.parent = this.svg.append("circle")
      .datum(this.root)
      .attr("r", this.radii(1)[0])
      .attr("fill", "none")
      .attr("pointer-events", "all")
      .style("cursor", "pointer")
      .on("click", (event, p) => this._clicked(event, p));

    this.labels();
    this.info();
  }

  _arcVisible(d) {
    return d.depth >= 1 && d.depth <= 3 && d.x1 > d.x0;
  }

  _clicked(event, p) {
    this._zoomDepth = p.depth;
    this.parent.datum(p.parent || this.root);

    this.root.each(d => {
      d.target = {
        x0: Math.max(0, Math.min(1, (d.original.x0 - p.original.x0) / (p.original.x1 - p.original.x0))) * 2 * Math.PI,
        x1: Math.max(0, Math.min(1, (d.original.x1 - p.original.x0) / (p.original.x1 - p.original.x0))) * 2 * Math.PI,
        depth: d.depth,
      };
    });

    this.text.selectAll("*").attr("opacity", 0);

    const duration = event.altKey ? 7500 : 750;
    const self = this;

    this.paths.transition().duration(duration)
      .tween("data", d => {
        const i = d3.interpolate(d.current, d.target);
        return t => d.current = i(t);
      })
      .filter(function(d) {
        return +this.getAttribute("fill-opacity") || self._arcVisible(d.target);
      })
      .attr("fill-opacity", d => {
        if (!this._arcVisible(d.target)) return 0;
        if (!this._selectedCountry) return 1;
        const names = d.ancestors().map(a => a.data.name);
        const tags = new Set(
          this.coffee
            .filter(r => r.origin === this._selectedCountry)
            .map(r => r.tag)
            .filter(Boolean)
        );
        return names.some(n => tags.has(n)) ? 1 : 0.1;
      })
      .attr("pointer-events", d => this._arcVisible(d.target) ? "auto" : "none")
      .attrTween("d", d => () => this.arc(d.current));

    d3.timeout(() => {
      this.text.selectAll("*").remove();
      this.labels();
    }, duration);
  }

  labels() {
    const zd    = this._zoomDepth ?? 0;
    const scale = 1 + zd * 0.35;

    const nodes = this.root.descendants().filter(d => d.depth && this._arcVisible(d.current));
    const arcNodes    = nodes.filter(d => d.depth === 1 || (d.depth === 2 && d.children));
    const radialNodes = nodes.filter(d => d.depth === 3 || (d.depth === 2 && !d.children));

    this.text.selectAll("path.label-path")
      .data(arcNodes)
      .join("path")
        .attr("class", "label-path")
        .attr("id", (d, i) => `arc-label-${i}`)
        .attr("fill", "none")
        .attr("d", d => {
          const [r0, r1] = this.radii(d.depth);
          const mid      = (r0 + r1) / 2;
          const cur      = d.current;
          const a0       = cur.x0 + chart.rings.gap / 2;
          const a1       = cur.x1 - chart.rings.gap / 2;
          const midAngle = (a0 + a1) / 2;

          const flipped = midAngle > Math.PI / 2 && midAngle < 3 * Math.PI / 2;
          const start   = flipped ? a1 : a0;
          const end     = flipped ? a0 : a1;

          const x1 = Math.sin(start) * mid, y1 = -Math.cos(start) * mid;
          const x2 = Math.sin(end)   * mid, y2 = -Math.cos(end)   * mid;
          const large = Math.abs(end - start) > Math.PI ? 1 : 0;
          const sweep = flipped ? 0 : 1;

          return `M ${x1},${y1} A ${mid},${mid} 0 ${large},${sweep} ${x2},${y2}`;
        });

    this.text.selectAll("text.arc-label")
      .data(arcNodes)
      .join("text")
        .attr("class", "arc-label")
        .attr("dy", "0.35em")
        .append("textPath")
          .attr("href", (d, i) => `#arc-label-${i}`)
          .attr("startOffset", "50%")
          .attr("text-anchor", "middle")
          .attr("fill", "#ffffffcc")
          .attr("font-size", d => d.depth === 1 ? `${12 * scale}px` : `${11 * scale}px`)
          .text(d => d.data.name);

    this.text.selectAll("text.radial-label")
      .data(radialNodes)
      .join("text")
        .attr("class", "radial-label")
        .attr("dy", "0.35em")
        .attr("text-anchor", d => {
          const mid = (d.current.x0 + d.current.x1) / 2;
          return mid < Math.PI ? "start" : "end";
        })
        .attr("fill", "#ffffff99")
        .attr("font-size", `${10 * scale}px`)
        .attr("transform", d => {
          const [, r1] = this.radii(d.depth);
          const mid    = (d.current.x0 + d.current.x1) / 2;
          const r      = r1 + 6;
          const x      = Math.sin(mid) * r;
          const y      = -Math.cos(mid) * r;
          let angle    = (mid * 180 / Math.PI) - 90;
          if (mid > Math.PI) angle += 180;
          return `translate(${x},${y}) rotate(${angle})`;
        })
        .text(d => d.data.name);
  }

  filter(country) {
    this._selectedCountry = country;
    if (!this.paths) return;
    if (!country) {
      this.paths.attr("fill-opacity", 1);
      return;
    }
    const tags = new Set(
      this.coffee
        .filter(d => d.origin === country)
        .map(d => d.tag)
        .filter(Boolean)
    );
    this.paths.attr("fill-opacity", d => {
      const names = d.ancestors().map(a => a.data.name);
      return names.some(n => tags.has(n)) ? 1 : 0.1;
    });
  }

  reset() {
    this._zoomDepth = 0; // ← fix: reset before labels() runs
    this.root.each(d => {
      d.target = {
        x0: d.original.x0,
        x1: d.original.x1,
        depth: d.depth,
      };
    });

    this.text.selectAll("*").attr("opacity", 0);

    this.paths.transition().duration(750)
      .tween("data", d => {
        const i = d3.interpolate(d.current, d.target);
        return t => d.current = i(t);
      })
      .attr("pointer-events", "auto")
      .attrTween("d", d => () => this.arc(d.current));

    d3.timeout(() => {
      this.text.selectAll("*").remove();
      this.labels();
      if (this._selectedCountry) this.filter(this._selectedCountry);
    }, 750);
  }

  info() {
    this.paths.append("title")
      .text(d => d.ancestors().reverse().map(d => d.data.name).join(" > "));
  }
}
```

```js
class Flavors {
  constructor(data, svg, scaaColors) {
    this.allData    = data;
    this.svg        = svg;
    this.scaaColors = scaaColors;
    this.g          = svg.append("g");
  }

  // Build color map using level-2 colors, cascading down to leaves
  static colorMap(scaaData) {
    const map = new Map();
    for (const l1 of scaaData.children ?? []) {
      for (const l2 of l1.children ?? []) {
        const color = l2.colour ?? l1.colour;
        map.set(l2.name, color);
        for (const l3 of l2.children ?? []) {
          map.set(l3.name, color);
        }
      }
    }
    return map;
  }

  _rollup(country) {
    const filtered = country
      ? this.allData.filter(d => d.origin === country)
      : this.allData;

    const map = new Map();
    for (const d of filtered) {
      const key = `${d.source}||${d.target}`;
      map.set(key, (map.get(key) ?? 0) + d.value);
    }
    return Array.from(map, ([key, value]) => {
      const [source, target] = key.split("||");
      return { source, target, value };
    }).filter(d => this.scaaColors.has(d.source) && this.scaaColors.has(d.target));
  }

  _matrix(pairs) {
    const tags = Array.from(new Set(pairs.flatMap(d => [d.source, d.target]))).sort();
    const idx  = new Map(tags.map((t, i) => [t, i]));
    const n    = tags.length;
    const mat  = Array.from({ length: n }, () => new Array(n).fill(0));
    for (const { source, target, value } of pairs) {
      const i = idx.get(source), j = idx.get(target);
      mat[i][j] += value;
      mat[j][i] += value;
    }
    return { tags, matrix: mat };
  }

  render(country = null) {
    this.g.selectAll("*").remove();

    const pairs = this._rollup(country);
    if (!pairs.length) return;

    const { tags, matrix } = this._matrix(pairs);
    const radius = Math.min(chart.size[0], chart.size[1]) / 2 * 0.85;
    const inner  = radius * 0.75;

    const chord  = d3.chord().padAngle(0.02).sortSubgroups(d3.descending)(matrix);
    const arc    = d3.arc().innerRadius(inner).outerRadius(radius);
    const ribbon = d3.ribbon().radius(inner);
    const color  = tag => this.scaaColors.get(tag) ?? "#555";

    // Ribbons
    this.g.append("g")
      .attr("fill-opacity", 0.7)
      .selectAll("path")
      .data(chord)
      .join("path")
        .attr("d", ribbon)
        .attr("fill", d => color(tags[d.source.index]))
        .attr("stroke", d => d3.color(color(tags[d.source.index]))?.darker())
        .append("title")
          .text(d => `${tags[d.source.index]} ↔ ${tags[d.target.index]}: ${d.source.value + d.target.value}`);

    // Arcs
    const group = this.g.append("g")
      .selectAll("g")
      .data(chord.groups)
      .join("g");

    group.append("path")
      .attr("d", arc)
      .attr("fill", d => color(tags[d.index]))
      .attr("stroke", "#111");

    // Labels
    const labelRadius = d => {
      const mid = (d.startAngle + d.endAngle) / 2;
      const flip = mid > Math.PI / 2 && mid < 3 * Math.PI / 2;
    
      // Move flipped labels slightly inward
      return flip ? radius + 18 : radius + 6;
    };
    
    group.append("path")
      .attr("id", d => `flavor-arc-${d.index}`)
      .attr("d", d => {
        const mid = (d.startAngle + d.endAngle) / 2;
    
        // Flip labels on left half of circle
        const flip = mid > Math.PI / 2 && mid < 3 * Math.PI / 2;
    
        const start = flip ? d.endAngle : d.startAngle;
        const end   = flip ? d.startAngle : d.endAngle;
        const sweep = flip ? 0 : 1;
    
        const r = labelRadius(d);
    
        const a0 = start - Math.PI / 2;
        const a1 = end   - Math.PI / 2;
    
        const x0 = Math.cos(a0) * r;
        const y0 = Math.sin(a0) * r;
        const x1 = Math.cos(a1) * r;
        const y1 = Math.sin(a1) * r;
    
        const large = Math.abs(end - start) > Math.PI ? 1 : 0;
    
        return `M ${x0},${y0} A ${r},${r} 0 ${large} ${sweep} ${x1},${y1}`;
      })
      .attr("fill", "none")
      .attr("stroke", "none");
    
    group.append("text")
      .attr("fill", "#ffffffcc")
      .attr("font-size", 20)
      .append("textPath")
        .attr("href", d => `#flavor-arc-${d.index}`)
        .attr("startOffset", "50%")
        .attr("text-anchor", "middle")
        .text(d => tags[d.index]);
    
  }

  filter(country) {
    this.render(country);
  }
}
```

```js
    WITH unnested AS (
      SELECT 
        UNNEST(origin) AS origin,
        UNNEST(tastingtag) AS tag
      FROM coffee
    ),
    counts AS (
      SELECT origin, tag, COUNT(*) AS freq
      FROM unnested
      WHERE tag IS NOT NULL
        AND origin != '-'
      GROUP BY origin, tag
    ),
    totals AS (
      SELECT origin, SUM(freq) AS total 
      FROM counts 
      GROUP BY origin
    )
    SELECT 
      c.origin,
      c.tag,
      ROUND(c.freq::DOUBLE / t.total, 3) AS weight
    FROM counts c 
    JOIN totals t ON c.origin = t.origin
    ORDER BY c.origin, weight DESC;
  `);
}
```

```js
  const raw = await db.query(`
    WITH tags AS (
      SELECT id, UNNEST(tastingtag) AS tag
      FROM coffee
    ),
    origins AS (
      SELECT id, UNNEST(origin) AS origin
      FROM coffee
    )
    SELECT 
      a.tag AS source,
      b.tag AS target,
      o.origin AS origin,
      COUNT(*) AS value
    FROM tags a
    JOIN tags b ON a.id = b.id AND a.tag < b.tag
    JOIN origins o ON a.id = o.id
    WHERE a.tag IS NOT NULL AND b.tag IS NOT NULL
    GROUP BY a.tag, b.tag, o.origin
    ORDER BY value DESC
  `);

  return raw
```
